```ruby
chair, group =  gets.split.map(&:to_i)

group_array = []
group.times do
  group_array << gets.split.map(&:to_i)
end

chair_array = Array.new(chair, 0)

group_array.each do |n|
  people, start = n
  check_array = chair_array.dup

  people.times do |i|
    index = (start -1 + i) % chair
    if check_array[index] == 0
        check_array[index] = 1
    else
        check_array = nil
        break
    end
  end
  chair_array = check_array if check_array
end

p chair_array.count(1)

  
```

手こずったのは、参照コピーと値コピー

```ruby
check_array = chair_array  # 参照コピーになってしまう
```

Ruby では配列を代入すると同じオブジェクトになる

```ruby
chair_array = [0, 0, 0]
check_array = chair_array  # 同じオブジェクトを指している

check_array[0] = 1
p chair_array  # => [1, 0, 0] ← chair_arrayも変わってしまう！
```

`dup` を使うと独立したコピーが作れる。

```ruby
check_array = chair_array.dup  # 別オブジェクトとしてコピー
```
