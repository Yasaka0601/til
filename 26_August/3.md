len, wid, players = gets.split.map(&:to_i)
#p len >> 2
#p wid >> 2
#p players >> 2

cards = Array.new(len) { gets.split.map(&:to_i) }
#p cards >> [[1, 2], [2, 1]]

memo = gets.to_i
# p memo >> 4

answers = Array.new(memo) { gets.split.map(&:to_i) }
# p answer >> [[1, 1, 2, 1], [1, 1, 1, 2], [1, 1, 2, 2], [1, 2, 2, 1]]

player_array = Array.new(players, 0)
player_index = 0
#p player_array >> [0, 0]

answers.each do |answer|
    l_1, w_1, l_2, w_2 = answer
    #p l_1 p w_1 p l_2 p w_2
    index = player_index % players
    #p index
    if cards[l_1 -1][w_1 -1] == cards[l_2 - 1][w_2 -1]
        player_array[index] += 2
    else
        player_index += 1
    end
end

puts player_array
