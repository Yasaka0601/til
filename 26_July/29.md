AI に質問後の修正

```ruby
chaer = gets.to_i
array = Array.new(chaer, 0)
a,b = gets.split.map(&:to_i)
a_array = (b...b+a).to_a

a.times do |i|
    b_count = b-1+i
    array_max = array.length - 1
    array[b_count % chaer] = a_array[i]
end

puts (0...a).map { |i| (b - 1 + i) % chaer + 1}.join(" ")
```
