```ruby
#　paiza君
# １回目２回目、１
# ３回目、(2-1回目のダメージ)= 1 + (2-2回目のダメージ)=0 == 1
# ４回目、(3-1回目のダメージ)= 1 + (3-2回目のダメージ)=1 == 2

# モンスター
# １回目２回目、１
# ３回目、(3-1回目のダメージ)*2 = 2 + (3-2回目のダメージ) = 1 == 3
# ４回目、(4-1回目のダメージ)*2 = 2 + (4-2回目のダメージ) = 1 == 3

paiza =  gets.to_i
paiza_attack = []
monster_attack = []
result = 0

while 0 < paiza do
    ##### paizaの攻撃 #####
    if  result >= 2
        paiza_attack << (monster_attack[result-1] + monster_attack[result-2])
    else # paiza_attack.length 2以下の攻撃
        paiza_attack << 1
    end
    
    ##### monsterの攻撃 #####
    if result >= 2
        monster_attack << (paiza_attack[result-1]*2 + paiza_attack[result-2])
    else # monster_attack.length 2以下の攻撃
        monster_attack << 1
    end
    paiza -= monster_attack[result]
    result += 1
end

puts result
```
